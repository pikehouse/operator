"""
CLI commands for the AI diagnosis agent.

Commands:
- operator agent start: Run the agent daemon
- operator agent diagnose: Diagnose a single ticket
"""

import asyncio
from pathlib import Path
from typing import Optional

import typer
from rich.console import Console

# Direct submodule imports to avoid circular import issues (per STATE.md decision)
from operator_core.agent.runner import AgentRunner

app = typer.Typer(help="AI diagnosis agent commands")
console = Console()


@app.command()
def start(
    interval: float = typer.Option(10.0, "--interval", "-i", help="Poll interval in seconds"),
    db_path: Path = typer.Option(
        Path.home() / ".operator" / "tickets.db",
        "--db",
        help="Path to tickets database",
    ),
    pd_endpoint: str = typer.Option(
        "http://localhost:2379",
        "--pd",
        envvar="PD_ENDPOINT",
        help="PD API endpoint",
    ),
    prometheus_url: str = typer.Option(
        "http://localhost:9090",
        "--prometheus",
        envvar="PROMETHEUS_URL",
        help="Prometheus URL",
    ),
    model: str = typer.Option(
        "claude-sonnet-4-5",
        "--model",
        "-m",
        help="Claude model for diagnosis",
    ),
) -> None:
    """
    Start the AI diagnosis agent daemon.

    Polls for undiagnosed tickets and invokes Claude for analysis.
    Runs until interrupted with Ctrl+C.

    Requires ANTHROPIC_API_KEY environment variable.
    """
    import httpx
    from operator_tikv.pd_client import PDClient
    from operator_tikv.prom_client import PrometheusClient
    from operator_tikv.subject import TiKVSubject

    async def run_agent():
        async with httpx.AsyncClient(base_url=pd_endpoint) as pd_http:
            async with httpx.AsyncClient(base_url=prometheus_url) as prom_http:
                subject = TiKVSubject(
                    pd=PDClient(http=pd_http),
                    prom=PrometheusClient(http=prom_http),
                )
                runner = AgentRunner(
                    subject=subject,
                    db_path=db_path,
                    poll_interval=interval,
                    model=model,
                )
                await runner.run()

    console.print(f"[bold green]Starting agent daemon[/]")
    console.print(f"  Poll interval: {interval}s")
    console.print(f"  Database: {db_path}")
    console.print(f"  Model: {model}")
    console.print()

    try:
        asyncio.run(run_agent())
    except KeyboardInterrupt:
        pass  # Graceful shutdown handled by AgentRunner


@app.command()
def diagnose(
    ticket_id: int = typer.Argument(..., help="Ticket ID to diagnose"),
    db_path: Path = typer.Option(
        Path.home() / ".operator" / "tickets.db",
        "--db",
        help="Path to tickets database",
    ),
    pd_endpoint: str = typer.Option(
        "http://localhost:2379",
        "--pd",
        envvar="PD_ENDPOINT",
        help="PD API endpoint",
    ),
    prometheus_url: str = typer.Option(
        "http://localhost:9090",
        "--prometheus",
        envvar="PROMETHEUS_URL",
        help="Prometheus URL",
    ),
    model: str = typer.Option(
        "claude-sonnet-4-5",
        "--model",
        "-m",
        help="Claude model for diagnosis",
    ),
) -> None:
    """
    Diagnose a single ticket immediately.

    Useful for testing or manual diagnosis of specific tickets.
    """
    import httpx
    from anthropic import AsyncAnthropic
    from operator_core.db.tickets import TicketDB
    from operator_core.agent.context import ContextGatherer
    from operator_core.agent.diagnosis import DiagnosisOutput, format_diagnosis_markdown
    from operator_core.agent.prompt import SYSTEM_PROMPT, build_diagnosis_prompt
    from operator_tikv.pd_client import PDClient
    from operator_tikv.prom_client import PrometheusClient
    from operator_tikv.subject import TiKVSubject

    async def run_diagnosis():
        async with TicketDB(db_path) as db:
            # Fetch the ticket
            ticket = await db.get_ticket(ticket_id)
            if not ticket:
                console.print(f"[red]Ticket {ticket_id} not found[/]")
                raise typer.Exit(1)

            if ticket.diagnosis:
                console.print(f"[yellow]Ticket {ticket_id} already diagnosed[/]")
                raise typer.Exit(0)

            async with httpx.AsyncClient(base_url=pd_endpoint) as pd_http:
                async with httpx.AsyncClient(base_url=prometheus_url) as prom_http:
                    subject = TiKVSubject(
                        pd=PDClient(http=pd_http),
                        prom=PrometheusClient(http=prom_http),
                    )

                    # Gather context
                    gatherer = ContextGatherer(subject, db)
                    context = await gatherer.gather(ticket)
                    prompt = build_diagnosis_prompt(context)

                    # Invoke Claude
                    client = AsyncAnthropic()
                    console.print(f"[dim]Invoking Claude ({model})...[/]")

                    response = await client.beta.messages.parse(
                        model=model,
                        max_tokens=4096,
                        betas=["structured-outputs-2025-11-13"],
                        system=SYSTEM_PROMPT,
                        messages=[{"role": "user", "content": prompt}],
                        output_format=DiagnosisOutput,
                    )

                    if response.stop_reason == "refusal":
                        console.print("[red]Claude refused to diagnose[/]")
                        raise typer.Exit(1)

                    # Format and store diagnosis
                    diagnosis_output = response.parsed_output
                    diagnosis_md = format_diagnosis_markdown(diagnosis_output)
                    await db.update_diagnosis(ticket_id, diagnosis_md)

                    console.print(f"[green]Ticket {ticket_id} diagnosed (severity: {diagnosis_output.severity})[/]")

    console.print(f"[bold]Diagnosing ticket {ticket_id}[/]")
    try:
        asyncio.run(run_diagnosis())
    except KeyboardInterrupt:
        console.print("[yellow]Cancelled[/]")
```

4. Diagnose command creates ContextGatherer, invokes Claude directly, stores result

@app.command()
def show(
    ticket_id: int = typer.Argument(..., help="Ticket ID to display"),
    db_path: Path = typer.Option(...),
) -> None:
    """Show detailed ticket information including diagnosis."""

    async def get_ticket():
        async with TicketDB(db_path) as db:
            return await db.get_ticket(ticket_id)

    ticket = asyncio.run(get_ticket())

    if not ticket:
        console.print(f"[red]Ticket {ticket_id} not found[/]")
        raise typer.Exit(1)

    # Display ticket metadata
    console.print(f"[bold]Ticket #{ticket.id}[/]")
    console.print(f"  Invariant: {ticket.invariant_name}")
    console.print(f"  Store: {ticket.store_id or 'cluster-wide'}")
    console.print(f"  Status: {ticket.status.value}")
    console.print(f"  Severity: {ticket.severity}")
    console.print(f"  First seen: {ticket.first_seen_at}")
    console.print(f"  Occurrences: {ticket.occurrence_count}")
    console.print()

    # Display diagnosis if present
    if ticket.diagnosis:
        console.print("[bold]Diagnosis:[/]")
        console.print()
        from rich.markdown import Markdown
        console.print(Markdown(ticket.diagnosis))
    else:
        console.print("[dim]No diagnosis yet[/]")
```
