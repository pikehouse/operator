from operator_core.monitor.types import Ticket
from operator_core.types import Store, ClusterMetrics

@dataclass
class DiagnosisContext:
    """All context needed for diagnosing a ticket."""
    ticket: Ticket
    metric_snapshot: dict[str, any]  # From ticket.metric_snapshot
    stores: list[Store]              # Current cluster topology
    cluster_metrics: ClusterMetrics  # Store/region counts, leader distribution
    log_tail: str | None             # Last N lines from affected component
    similar_tickets: list[Ticket]    # Past diagnoses for same invariant
```

class ContextGatherer:
    """Assembles diagnosis context from multiple sources.

    Per CONTEXT.md decisions:
    - Metric snapshot only (current values at violation time)
    - Include raw log tail (last N lines)
    - Include similar ticket history (past diagnoses for pattern matching)
    - Include full cluster topology
    """

    def __init__(
        self,
        subject: TiKVSubject,
        db: TicketDB,
        log_tail_lines: int = 50,
        similar_ticket_days: int = 7,
    ):
        self.subject = subject
        self.db = db
        self.log_tail_lines = log_tail_lines
        self.similar_ticket_days = similar_ticket_days

    async def gather(self, ticket: Ticket) -> DiagnosisContext:
        """Gather all context for diagnosing a ticket."""
        # Get current cluster state
        stores = await self.subject.get_stores()
        cluster_metrics = await self.subject.get_cluster_metrics()

        # Get metric snapshot from ticket (already captured at violation time)
        metric_snapshot = ticket.metric_snapshot or {}

        # Fetch log tail for affected component (stub for now - returns None)
        log_tail = await self._fetch_log_tail(ticket.store_id)

        # Find similar tickets (same invariant, recent)
        similar_tickets = await self._find_similar_tickets(ticket)

        return DiagnosisContext(
            ticket=ticket,
            metric_snapshot=metric_snapshot,
            stores=stores,
            cluster_metrics=cluster_metrics,
            log_tail=log_tail,
            similar_tickets=similar_tickets,
        )
```

   - "Wait and observe" is a valid recommendation
